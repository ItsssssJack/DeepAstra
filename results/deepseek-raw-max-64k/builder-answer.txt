(function () {
  'use strict';

    });
  }

  /* ---------------------------------------------------------

  var logoButtons = Array.prototype.slice.call(document.querySelectorAll('[data-logo-pick]'));

  logoButtons.forEach(function (btn) {

      var svg = btn.querySelector('svg');

      document.querySelectorAll('[data-logo]').forEach(function (container) {
        container.innerHTML = svg.outerHTML;
      });
      logoButtons.forEach(function (other) {
        other.setAttribute('aria-pressed', String(other === btn));
      });
    });
  });

  /* ---------------------------------------------------------
     6. Demo page background
     --------------------------------------------------------- */
  var pageBgInput = document.getElementById('page-bg');
  var hexPattern = /^#([0-9a-f]{3}|[0-9a-f]{6})$/i;

  if (pageBgInput) {
    pageBgInput.addEventListener('input', function () {
      var value = pageBgInput.value.trim();
      if (hexPattern.test(value)) {

        pageBgInput.setAttribute('aria-invalid', 'true');

          errorEl.hidden = false;

        emailInput.focus();
        return;

      if (errorEl) { errorEl.hidden = true; }
      emailInput.removeAttribute('aria-invalid');

          noteEl.hidden = true;
        }, 3200);

  /* ---------------------------------------------------------

     --------------------------------------------------------- */
  var prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)');

  document.querySelectorAll('a[href^="#"]').forEach(function (link) {

      if (!target) { return; }

      target.scrollIntoView({

        block: 'start'
      });

      if (history.replaceState) {
        history.replaceState(null, '', id);
      }
    });
  });

### Visual Design Highlights
